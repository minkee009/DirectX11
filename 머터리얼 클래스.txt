
class Material
{
private:
    UINT m_textureFlags = 0; // 각 TextureType에 해당하는 bitmask
    ComPtr<ID3D11Buffer> m_materialCB; // 상수버퍼
public:
    void UpdateTextureFlags()
    {
        m_textureFlags = 0;
        for (auto& tex : textures)
        {
            if (tex.pSRV)
                m_textureFlags |= static_cast<UINT>(tex.type);
        }
    }
};

struct MaterialCB
{
    UINT textureFlags; // bit0=Diffuse, bit1=Specular, bit2=Normal, ...
    float padding[3];  // 16byte align
};

#----- HLSL 코드 -----#
void Material::Bind(ID3D11DeviceContext* context)
{
    // 1. 셰이더 바인딩
    if (m_vertexShader) context->VSSetShader(m_vertexShader.Get(), nullptr, 0);
    if (m_pixelShader)  context->PSSetShader(m_pixelShader.Get(), nullptr, 0);

    // 2. 텍스처 바인딩 + Sampler
    for (auto& tex : textures)
    {
        ID3D11ShaderResourceView* srv = tex.pSRV.Get();
        context->PSSetShaderResources(tex.slot, 1, srv ? &srv : nullptr);
        ID3D11SamplerState* sampler = tex.pSampler.Get();
        context->PSSetSamplers(tex.slot, 1, sampler ? &sampler : nullptr);
    }

    // 3. Material 상수버퍼에 텍스처 bitmask 전달
    UpdateTextureFlags(); // 현재 존재하는 텍스처 기반 bitmask 계산

    MaterialCB cb;
    cb.textureFlags = m_textureFlags;

    context->UpdateSubresource(m_materialCB.Get(), 0, nullptr, &cb, 0, 0);
    context->PSSetConstantBuffers(0, 1, m_materialCB.GetAddressOf());
}
