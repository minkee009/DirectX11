//=== Vertex 채널 정의 ===//
struct VertexChannels
{
    std::vector<XMFLOAT3> positions;
    std::vector<XMFLOAT3> normals;
    std::vector<XMFLOAT2> uvs;
};

//=== Vertex 채널 플래그 ===//
enum class VertexChannelFlag : uint8_t
{
    None   = 0,
    Pos    = 1 << 0,
    Normal = 1 << 1,
    UV     = 1 << 2
};
inline VertexChannelFlag operator|(VertexChannelFlag a, VertexChannelFlag b)
{
    return static_cast<VertexChannelFlag>(static_cast<uint8_t>(a) | static_cast<uint8_t>(b));
}
inline bool HasFlag(VertexChannelFlag flags, VertexChannelFlag test)
{
    return (static_cast<uint8_t>(flags) & static_cast<uint8_t>(test)) != 0;
}

//=== 간단한 Mesh 구조 ===//
struct Mesh
{
    VertexChannels channels;
    std::vector<UINT> indices;
    ComPtr<ID3D11Buffer> vertexBuffer;
    ComPtr<ID3D11Buffer> indexBuffer;
    VertexChannelFlag channelFlags = VertexChannelFlag::Pos | VertexChannelFlag::Normal | VertexChannelFlag::UV;

    void CreateBuffers(ID3D11Device* device)
    {
        // Vertex Buffer 생성
        std::vector<uint8_t> vertexData;
        size_t vertexCount = channels.positions.size();

        for (size_t i = 0; i < vertexCount; ++i)
        {
            if (HasFlag(channelFlags, VertexChannelFlag::Pos))
                vertexData.insert(vertexData.end(), (uint8_t*)&channels.positions[i], (uint8_t*)&channels.positions[i] + sizeof(XMFLOAT3));
            if (HasFlag(channelFlags, VertexChannelFlag::Normal))
                vertexData.insert(vertexData.end(), (uint8_t*)&channels.normals[i], (uint8_t*)&channels.normals[i] + sizeof(XMFLOAT3));
            if (HasFlag(channelFlags, VertexChannelFlag::UV))
                vertexData.insert(vertexData.end(), (uint8_t*)&channels.uvs[i], (uint8_t*)&channels.uvs[i] + sizeof(XMFLOAT2));
        }

        D3D11_BUFFER_DESC vbd{};
        vbd.Usage = D3D11_USAGE_DEFAULT;
        vbd.BindFlags = D3D11_BIND_VERTEX_BUFFER;
        vbd.ByteWidth = static_cast<UINT>(vertexData.size());
        D3D11_SUBRESOURCE_DATA initData{};
        initData.pSysMem = vertexData.data();
        device->CreateBuffer(&vbd, &initData, vertexBuffer.GetAddressOf());

        // Index Buffer 생성
        if (!indices.empty())
        {
            D3D11_BUFFER_DESC ibd{};
            ibd.Usage = D3D11_USAGE_DEFAULT;
            ibd.BindFlags = D3D11_BIND_INDEX_BUFFER;
            ibd.ByteWidth = static_cast<UINT>(indices.size() * sizeof(UINT));
            D3D11_SUBRESOURCE_DATA idata{};
            idata.pSysMem = indices.data();
            device->CreateBuffer(&ibd, &idata, indexBuffer.GetAddressOf());
        }
    }
};

//=== Input Layout 생성 예제 ===//
ComPtr<ID3D11InputLayout> CreateInputLayout(ID3D11Device* device, ID3DBlob* vsBlob, VertexChannelFlag flags)
{
    std::vector<D3D11_INPUT_ELEMENT_DESC> layoutDesc;
    UINT offset = 0;

    if (HasFlag(flags, VertexChannelFlag::Pos))
    {
        layoutDesc.push_back({ "POSITION", 0, DXGI_FORMAT_R32G32B32_FLOAT, 0, offset, D3D11_INPUT_PER_VERTEX_DATA, 0 });
        offset += sizeof(XMFLOAT3);
    }
    if (HasFlag(flags, VertexChannelFlag::Normal))
    {
        layoutDesc.push_back({ "NORMAL", 0, DXGI_FORMAT_R32G32B32_FLOAT, 0, offset, D3D11_INPUT_PER_VERTEX_DATA, 0 });
        offset += sizeof(XMFLOAT3);
    }
    if (HasFlag(flags, VertexChannelFlag::UV))
    {
        layoutDesc.push_back({ "TEXCOORD", 0, DXGI_FORMAT_R32G32_FLOAT, 0, offset, D3D11_INPUT_PER_VERTEX_DATA, 0 });
        offset += sizeof(XMFLOAT2);
    }

    ComPtr<ID3D11InputLayout> inputLayout;
    device->CreateInputLayout(layoutDesc.data(), (UINT)layoutDesc.size(), vsBlob->GetBufferPointer(), vsBlob->GetBufferSize(), inputLayout.GetAddressOf());
    return inputLayout;
}

//=== Renderer에서 Draw 호출 ===//
void RenderMesh(ID3D11DeviceContext* ctx, Mesh& mesh, ID3D11InputLayout* inputLayout)
{
    UINT stride = 0;
    UINT offset = 0;

    // stride 계산
    stride += HasFlag(mesh.channelFlags, VertexChannelFlag::Pos) ? sizeof(XMFLOAT3) : 0;
    stride += HasFlag(mesh.channelFlags, VertexChannelFlag::Normal) ? sizeof(XMFLOAT3) : 0;
    stride += HasFlag(mesh.channelFlags, VertexChannelFlag::UV) ? sizeof(XMFLOAT2) : 0;

    ctx->IASetInputLayout(inputLayout.Get());
    ctx->IASetVertexBuffers(0, 1, mesh.vertexBuffer.GetAddressOf(), &stride, &offset);
    ctx->IASetIndexBuffer(mesh.indexBuffer.Get(), DXGI_FORMAT_R32_UINT, 0);
    ctx->IASetPrimitiveTopology(D3D11_PRIMITIVE_TOPOLOGY_TRIANGLELIST);

    ctx->DrawIndexed((UINT)mesh.indices.size(), 0, 0);
}
